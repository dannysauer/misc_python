Outage, reversal and cold-night simulations of the setpoint-pair control,
captured 2026-11-17.

These back the tables in docs/planning/follow-up.md that the closed-loop
receipt of 2026-11-16 does not cover. All three build on tests/closed_loop.py,
which describes the simulated house, thermostat and schedule.

The outage and reversal scenarios are scripts that reviewing agents wrote.
They are not part of the test suite. Their text is at the end of this file as
it was run, apart from the path to the tests directory, so that they can be
run again from the project directory with: uv run python <script>

Script for section 2:

"""The closed-loop house with an ecobee outage: for a while Home Assistant accepts
climate.set_temperature (pyecobee logs the failure and returns) and nothing reaches
the thermostat, whose last known state Home Assistant goes on reporting.

Run from the project directory: uv run python <this file>
"""

import sys
from pathlib import Path
from tempfile import TemporaryDirectory

sys.path.insert(0, "tests")

from closed_loop import (  # noqa: E402
    DIFFERENTIAL,
    STEPS_BETWEEN_RUNS,
    STEPS_PER_DAY,
    STEPS_PER_HOUR,
    TARGETS,
    SimulatedHome,
    _config,
    trained_house,
)
from fakes import FakeMetrics  # noqa: E402

from hvac_predict.advisor import Steered, steer  # noqa: E402
from hvac_predict.decision import HEAT, Goal  # noqa: E402
from hvac_predict.training import save_model  # noqa: E402


class OutageHome(SimulatedHome):
    def __init__(self, simulation, outages, indefinite):
        super().__init__(simulation)
        self.outages = outages  # list of (first index, last index)
        self.lost = 0

    def down(self):
        index = self._simulation.index
        return any(first <= index < last for first, last in self.outages)

    def set_temperature_range(self, entity_id, low, high):
        if self.down():
            self.lost += 1
            return
        super().set_temperature_range(entity_id, low, high)


def run(days, mean_outdoor, outage_minutes, every_hours, *, indefinite):
    simulation, model = trained_house(mean_outdoor)
    simulation.setpoints = Goal(TARGETS[simulation.comfort], DIFFERENTIAL).pair(HEAT)
    start = simulation.index
    outage_steps = outage_minutes // 5
    outages = [
        (at, at + outage_steps)
        for at in range(start + 7, start + days * STEPS_PER_DAY, every_hours * STEPS_PER_HOUR)
    ]
    home = OutageHome(simulation, outages, indefinite)
    left_alone_runs = 0
    episodes = 0  # outages after which the tool was left disowning its own hold
    previous_left = False
    with TemporaryDirectory() as directory:
        config = _config(Path(directory), {})
        save_model(model, config.model_path)
        for _ in range(days * STEPS_PER_DAY):
            if indefinite:
                # A hold that lasts until changed: the schedule does not take the setpoints back.
                pass
            else:
                simulation.follow_schedule(HEAT)
            simulation.read()
            if simulation.index % STEPS_BETWEEN_RUNS == 0:
                metrics = FakeMetrics(simulation.history(), end=simulation.now)
                outcome = steer(config, metrics, home, now=simulation.now, apply=True)
                left = not isinstance(outcome, Steered)
                left_alone_runs += left
                episodes += left and not previous_left
                previous_left = left
            simulation.advance()
    return len(outages), home.lost, episodes, left_alone_runs


for indefinite in (False, True):
    kind = "holds last until changed" if indefinite else "holds end at the next comfort change"
    for minutes in (10, 20, 30, 60):
        outages, lost, episodes, runs = run(6, 14.0, minutes, 5, indefinite=indefinite)
        print(
            f"{kind}; {outages} outages of {minutes} min: {lost} sends lost, tool disowned its "
            f"own hold {episodes} times, {runs} of 864 runs left the thermostat alone "
            f"({runs / 6:.0f} h)"
        )


Script for section 3:

"""How often does the preference the tool sends reverse, run to run, in the closed loop?

Run from the project directory: uv run python <this file>
"""

import sys
from collections import Counter
from pathlib import Path
from tempfile import TemporaryDirectory

sys.path.insert(0, "tests")

from closed_loop import (  # noqa: E402
    DIFFERENTIAL,
    STEPS_BETWEEN_RUNS,
    STEPS_PER_DAY,
    TARGETS,
    SimulatedHome,
    _config,
    trained_house,
)
from fakes import FakeMetrics  # noqa: E402

from hvac_predict.advisor import Steered, steer  # noqa: E402
from hvac_predict.decision import HEAT, Goal  # noqa: E402
from hvac_predict.training import save_model  # noqa: E402


def run(days, mean_outdoor, **decision):
    simulation, model = trained_house(mean_outdoor)
    simulation.setpoints = Goal(TARGETS[simulation.comfort], DIFFERENTIAL).pair(HEAT)
    home = SimulatedHome(simulation)
    log = []  # (index, comfort, preference sent, rule, applied, left_alone)
    with TemporaryDirectory() as directory:
        config = _config(Path(directory), decision)
        save_model(model, config.model_path)
        for _ in range(days * STEPS_PER_DAY):
            simulation.follow_schedule(HEAT)
            simulation.read()
            if simulation.index % STEPS_BETWEEN_RUNS == 0:
                metrics = FakeMetrics(simulation.history(), end=simulation.now)
                outcome = steer(config, metrics, home, now=simulation.now, apply=True)
                if isinstance(outcome, Steered):
                    made = outcome.advice.decision
                    log.append((simulation.index, simulation.comfort, made.preference, made.rule, outcome.applied))
                else:
                    log.append((simulation.index, simulation.comfort, None, "LEFT ALONE", False))
            simulation.advance()
    return log, home.changes


def summarise(log):
    # A reversal: this run's preference differs from the last run's, within one comfort setting.
    reversals = [
        (a, b) for a, b in zip(log, log[1:]) if a[1] == b[1] and a[2] != b[2]
    ]
    kinds = Counter(f"{a[3]}->{b[3]}" for a, b in reversals)
    # Longest streak of consecutive runs that each reverse the one before.
    longest = streak = 0
    for a, b in zip(log, log[1:]):
        streak = streak + 1 if (a[1] == b[1] and a[2] != b[2]) else 0
        longest = max(longest, streak)
    # Most reversals inside any hour (6 runs).
    flags = [a[1] == b[1] and a[2] != b[2] for a, b in zip(log, log[1:])]
    most = max(sum(flags[i : i + 6]) for i in range(len(flags)))
    left = sum(entry[3] == "LEFT ALONE" for entry in log)
    return len(reversals), longest, most, dict(kinds), left


for mean in (2.0, 10.0, 14.0, 18.0, 22.0):
    for settings in (
        {},
        {"overshoot_minutes": 90.0},
        {"overshoot_minutes": 30.0},
        {"min_hold_minutes": 120.0},
    ):
        log, changes = run(6, mean, **settings)
        count, longest, most, kinds, left = summarise(log)
        print(
            f"mean {mean:4.1f} {settings!s:32} runs {len(log)} changes {changes:3d} reversals {count:3d} "
            f"longest run-after-run streak {longest} most in an hour {most} left alone {left}"
        )
        print(f"      {kinds}")
